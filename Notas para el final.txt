Cosas que se podrian mejorar:
 - Usar un patron builder para limpiar esa lógica repetitiva de "si el valor viene vacío, dejo el viejo; si viene nuevo, lo cambio"
   Porque esto relacionado a lo poo va por el lado del encapsulamiento... La regla de negocio "Si el input es 0, ignóralo" ahora vive dentro del Builder, no "sucia" en el medio del Gestor
   En resumen pasariamos de pasasariamos de (marca == "") ? vehiculoOriginal.Marca : marca (repetido 6 veces) a .ConMarca(marca).
   Habría que ahcer una clase base abstracta para manejar las propiedades comunes (Marca, Modelo, Precio, etc.) y la lógica de validación (si es "" o 0, no actualiza)
   Y luego las clases especificas que heredan de la base y añaden sus propiedades únicas.

	Ejemplo:
	public abstract class VehiculoBuilder<T> where T : VehiculoBuilder<T>
    {
        protected string _marca;
        protected string _modelo;
        protected int _patentamiento;
        protected decimal _precio;
        protected int _cilindrada;

        // El constructor recibe el vehículo original para "clonar" sus datos iniciales
        protected VehiculoBuilder(Vehiculo original)
        {
            _marca = original.Marca;
            _modelo = original.Modelo;
            _patentamiento = original.Patentamiento;
            _precio = original.Precio;
            _cilindrada = original.Cilindrada;
        }

        // Métodos "Fluent" (devuelven 'this' para encadenar llamadas)
        public T ConMarca(string marca)
        {
            if (!string.IsNullOrWhiteSpace(marca)) _marca = marca;
            return (T)this;
        }



public class AutoBuilder : VehiculoBuilder<AutoBuilder>
{
    private int _puertas;

    // Inicializamos con los datos del Auto original
    public AutoBuilder(Auto original) : base(original)
    {
        _puertas = original.CantidadPuertas;
    }

    public AutoBuilder ConPuertas(int puertas)
    {
        if (puertas > 0) _puertas = puertas;
        return this;
    }

    public override Vehiculo Build()
    {
        // Aquí se llama al constructor real, limpio y ordenado
        return new Auto(_marca, _modelo, _patentamiento, _precio, _cilindrada, _puertas);
    }
